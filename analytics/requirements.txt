# Dashboard-only dependencies for the hosted demo (Streamlit Community Cloud).
#
# Community Cloud searches the entrypoint's directory before the repo root, so
# this file takes precedence over the root requirements.txt for the deployed app.
#
# The root requirements.txt pins Spark, Airflow and Iceberg —
# the production pipeline stack. None of it is needed to serve the dashboard,
# which runs the pure-Pandas scoring engines from pipeline/aggregation/.
# Installing the full set on a hosted runner exceeds its build limits.
#
# Every pin below ships a cp314 wheel. Community Cloud builds on whatever
# Python it defaults to — currently 3.14 — and the previous pins (streamlit
# 1.35 / pandas 2.2.2 / numpy 1.26.4) had no 3.14 wheels, so pip fell back to
# compiling from source and the deploy failed on Pillow: "The headers or
# library files could not be found for zlib". Verified by installing this file
# on Python 3.14.7: all wheels, no source builds.
#
# Floor, not just a ceiling: numpy 2.3 and scipy 1.16 require Python >= 3.11,
# which analytics/Dockerfile (python:3.11-slim) satisfies.

streamlit==1.64.0
plotly==6.9.0
pandas==2.3.3
numpy==2.3.2
scipy==1.16.1
click==8.5.0
loguru==0.7.3

# Required by plotly.express trendline="ols" on the strain-vs-recovery
# scatter. Without it the dashboard raises ModuleNotFoundError mid-render.
statsmodels==0.14.5
